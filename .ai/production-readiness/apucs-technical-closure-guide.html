Session 3.1 — Register interchangeable backends

Code⌄
Time45m
Tokens~10K
Backend registry 
class BackendRegistry {
  constructor() { this.backends = new Map(); }
  register(id, backend) { this.backends.set(id, backend); }
  get(id) {
    const backend = this.backends.get(id);
    if (!backend) throw new Error(`Unknown backend: ${id}`);
    return backend;
  }
}

module.exports = { BackendRegistry };

Session 3.2 — Complete hard constraints and selection

Code⌄
Time90m
Tokens~20K
Selection gate 
function hardGate(candidate) {
  return candidate.anchorRecall === 1
    && candidate.actionRecall === 1
    && candidate.provenanceCoverage === 1
    && candidate.contradictions === 0
    && candidate.adviceViolations === 0;
}

function selectCandidate(candidates) {
  const eligible = candidates.filter(hardGate);
  return eligible.sort((a, b) => a.totalCost - b.totalCost)[0] || { state: 'REVIEW' };
}
Close: JS divergence/demand gap remains a soft selection term; hard safety checks cannot be bypassed.

Session 4.1 — Language detection and normalized values

Locale normalization 
function normalizeAmount(value, language) {
  const raw = String(value).replace(/[^0-9,.-]/g, '');
  if (language === 'es') return Number(raw.replace(/\./g, '').replace(',', '.'));
  return Number(raw.replace(/,/g, ''));
}

function languageProfile(language) {
  if (!['en', 'es'].includes(language)) throw new Error('Unsupported language');
  return require(`../../config/profiles/insurance/${language}.json`);
}

module.exports = { normalizeAmount, languageProfile };
Close: English and Spanish normalize dates, amounts, negation, conditions, and entities into the same downstream contract.

Three-outcome classifier contract 
function classifyAdvice(sentence, { language, domain }) {
  const result = model.predict({ sentence, language, domain });
  if (result.confidence < 0.7) return { state: 'REVIEW', confidence: result.confidence };
  return result.adviceLike
    ? { state: 'ADVICE_LIKE', confidence: result.confidence }
    : { state: 'SOURCE_GROUNDED', confidence: result.confidence };
}
